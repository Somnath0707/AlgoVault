            if(s.charAt(i) == t.charAt(i))continue; 
            if(s.charAt(i) == '0'){
                zeroF++;
            }else{
                oneF++; 
            }
        }

        long cost = 0 ; 

        int pairs = Math.min(zeroF , oneF); 

        cost += 1L * pairs * Math.min(swapCost , 2 * flipCost); 

        zeroF -= pairs; 
        oneF -= pairs; 

        int remain = (zeroF + oneF);

        pairs = remain /2 ; 

        cost += 1L * pairs * Math.min(maxCrossCost , 2 * flipCost); 

        if(remain % 2 == 1){
            cost += flipCost; 
        }

        return cost ; 
    }
