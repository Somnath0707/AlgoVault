et.contains(v)) continue;

                    int total = len + currLen;

                    if (total < dist[v]) {
                        dist[v] = total;
                        pq.offer(new int[]{v, total});
                    }
                }
            }

            for (int i = 0; i < n; i++) {

                if (set.contains(i)) continue;

                if (dist[i] > max) {
                    return false;
                }
            }
        }

        return true;
    }

    public int f(int i, Set<Integer> set, List<List<int[]>> list,
                 int maxDistance, int n) {

        if (i == n) {
            if(dijkstra(set,list,maxDistance , n)) return 1; 
            return 0; 
        }

        int take = 0;
        int skip = 0;

        set.add(i);
        take = f(i + 1, set, list, maxDistance, n);
        set.remove(i);

        skip = f(i + 1, set, list, maxDistance, n);

        return take + skip;
    }
